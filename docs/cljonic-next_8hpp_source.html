74struct RangeNextAccess {
75 template <concepts::CljonicRange P>
76 [[nodiscard]] static constexpr auto tail(const P& producer) noexcept -> P {
77 if (producer.step_ == 0) {

96struct CycleNextAccess {
97 template <concepts::CljonicCycle P>
98 [[nodiscard]] static constexpr auto finite_tail(const P& producer) noexcept -> P {
99 const auto source_count = producer.source_.count();
100 if (source_count == 0U) {
101 return producer;
102 }
103 const auto offset = producer.starting_offset_ % source_count;

389template <concepts_detail::NextSequenceableCollection C>
390[[nodiscard]] constexpr auto next(const C& collection) noexcept -> concepts_detail::NextResult<C>::type {
391 using Result = concepts_detail::NextResult<C>::type;
392 Result result{};
393 auto iterator = collection.begin();
394 const auto end = collection.end();
395 if (iterator != end) {
396 ++iterator;
397 }
398 for (; iterator != end; ++iterator) {
399 result = result.conj(*iterator);
400 }
401 return result;
402}

409template <concepts::CljonicCycle P>
410[[nodiscard]] constexpr auto concepts_detail::CycleNextAccess::tail(const P& producer) noexcept -> P {
411 if constexpr (CLJONIC_COLLECTION_MAXIMUM_ELEMENT_COUNT_VALUE == 0U) {
412 return producer;
413 }
414 if constexpr (concepts::CljonicCollection<typename P::source_type>) {
415 return finite_tail(producer);
