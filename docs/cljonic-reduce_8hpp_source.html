118// The observed element count is the source's count saturated at the collection
119// maximum, so an unbounded producer folds over its observation cap and the
120// operation always terminates (REQ-FN-034).
121template <typename Source>
122[[nodiscard]] constexpr auto reduce_observed_count(const Source& source) noexcept -> std::size_t {
123 return std::min<std::size_t>(static_cast<std::size_t>(source.count()),
124 CLJONIC_COLLECTION_MAXIMUM_ELEMENT_COUNT_VALUE);

127template <typename F, typename Iterator, typename Accumulator>
128[[nodiscard]] constexpr auto reduce_fold(const F& function, Accumulator accumulator, Iterator iterator,
129 std::size_t remaining) noexcept -> Accumulator {
130 for (std::size_t step = 0U; step < remaining; ++step) {
131 accumulator = std::invoke(function, accumulator, *iterator);
132 ++iterator;
133 }
134 return accumulator;
135}

229template <typename F, typename Source>
230 requires concepts_detail::ReduceCallable<F, Source>
231[[nodiscard]] constexpr auto reduce(const F& function, const Source& source) noexcept
232 -> concepts_detail::ReduceAccumulatorFromSignature<F>::type {
233 using Accumulator = concepts_detail::ReduceAccumulatorFromSignature<F>::type;
234
235 const auto observed = concepts_detail::reduce_observed_count(source);
236 if (observed == 0U) {
237 return Accumulator{};
238 }
239 auto iterator = std::ranges::begin(source);
240 auto accumulator = static_cast<Accumulator>(*iterator);
241 ++iterator;
242 return concepts_detail::reduce_fold(function, accumulator, iterator, observed - 1U);
243}
