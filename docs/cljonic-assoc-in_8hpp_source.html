98template <typename C, typename P, typename V, std::size_t PathCapacity>
99constexpr auto assoc_in_map_at(const C& collection, const cljonic::Vector<P, PathCapacity>& path, std::size_t index,
100 const V& value) -> C {
101 static_assert(concepts::SafeImplicitConversion<typename C::key_type, P>,
102 "cljonic::assoc_in: a Map path element must safely convert to the Map key type.");
