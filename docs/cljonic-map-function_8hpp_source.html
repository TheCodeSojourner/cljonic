65// Per-position parameter rule (REQ-FN-032): when the callback's parameter types
66// are statically known, each parameter must accept its corresponding source
67// element through a safe implicit conversion. Narrowing conversions and
68// Boolean/character/enumeration/pointer/user-defined crossings are rejected. A
69// callback with fewer parameters than sources is rejected; extra parameters may
70// take defaults.
71template <typename Params, typename IndexSequence, typename... Sources>
72struct MapParametersSafe;
73
74template <typename Params, std::size_t... Index, typename... Sources>
75struct MapParametersSafe<Params, std::index_sequence<Index...>, Sources...>
76 : std::bool_constant<(concepts::SafeImplicitConversion<std::tuple_element_t<Index, Params>,
77 std::ranges::range_value_t<const Sources>> &&
78 ...)> {};

80template <typename F, typename... Sources>
81inline constexpr bool map_parameters_safe_v = [] constexpr {
82 using Parameters = CallableParameters<std::decay_t<F>>::type;
83 if constexpr (!CallableParameters<std::decay_t<F>>::known) {
84 return true;
85 } else if constexpr (std::tuple_size_v<Parameters> < sizeof...(Sources)) {
86 return false;
87 } else {
88 return MapParametersSafe<Parameters, std::index_sequence_for<Sources...>, Sources...>::value;
89 }
90}();

92// Guarded admission for the multi-source `map` free function (REQ-FN-032):
93// every source must be a cljonic source, the transforming function must be
94// copy-constructible and invoked without throwing with one argument per
95// source, its result must be a nothrow-storable element, and -- when its
96// parameter types are inspectable -- every parameter must safely accept the
97// corresponding source element.
98template <typename F, typename... Sources>
99concept MapCallable =
100 (sizeof...(Sources) >= 1) && (concepts::CljonicSource<Sources> && ...) &&
101 std::copy_constructible<std::decay_t<F>> && std::is_nothrow_copy_constructible_v<std::decay_t<F>> &&
102 std::is_nothrow_invocable_v<const std::decay_t<F>&, std::ranges::range_reference_t<const Sources>...> &&
103 concepts::NothrowCollectionElement<
104 std::invoke_result_t<const std::decay_t<F>&, std::ranges::range_reference_t<const Sources>...>> &&
105 map_parameters_safe_v<F, Sources...>;
106
107template <typename F, typename... Sources>
108using MapResultType = std::invoke_result_t<const std::decay_t<F>&, std::ranges::range_reference_t<const Sources>...>;

110// Compile-time observable-count bound per source: a collection contributes its
111// declared capacity; any producer contributes the configured observable
112// traversal cap (REQ-FN-032 result-capacity policy).
113template <typename Source>
114[[nodiscard]] constexpr auto map_source_bound() noexcept -> std::size_t {
115 if constexpr (concepts::CljonicCollection<Source>) {
116 return Source::capacity();
117 } else {
118 return CLJONIC_COLLECTION_MAXIMUM_ELEMENT_COUNT_VALUE;
119 }
120}
121
122template <typename... Sources>
123inline constexpr std::size_t MapResultCapacity = [] constexpr {
124 std::size_t minimum = CLJONIC_COLLECTION_MAXIMUM_ELEMENT_COUNT_VALUE;
125 ((minimum = std::min(minimum, map_source_bound<Sources>())), ...);
126 return minimum;
127}();

210template <typename F, typename... Sources>
211 requires concepts_detail::MapCallable<F, Sources...>
212[[nodiscard]] constexpr auto map(const F& function, const Sources&... sources) noexcept
213 -> Vector<concepts_detail::MapResultType<F, Sources...>, concepts_detail::MapResultCapacity<Sources...>> {
214 constexpr auto capacity = concepts_detail::MapResultCapacity<Sources...>;
215 using Result = concepts_detail::MapResultType<F, Sources...>;
216
217 const std::size_t mapped_count =
218 std::min({static_cast<std::size_t>(sources.count())..., static_cast<std::size_t>(capacity)});
219
220 auto iterators = std::tuple{sources.begin()...};
221 Vector<Result, capacity> result{};
222 for (std::size_t index = 0U; index < mapped_count; ++index) {
223 const auto value =
224 std::apply([&function](const auto&... iterator) { return std::invoke(function, *iterator...); }, iterators);
225 result = result.conj(value);
226 std::apply([](auto&... iterator) { (++iterator, ...); }, iterators);
227 }
228 return result;
229}
