214// Convert an admitted API index only when its value is representable by the
215// library's normalized size_t index domain. Collection-specific bounds remain
216// at each collection's call site because lookup and association have different
217// valid ranges.
218template <IntegerIndexType IndexType>
219[[nodiscard]] constexpr auto try_normalize_index(IndexType index) noexcept -> std::optional<std::size_t> {
220 if (!std::in_range<std::size_t>(index)) {
221 return std::nullopt;
222 }
223 return static_cast<std::size_t>(index);

335// Non-throwing equality of a type's own comparison (REQ-FN-002G): the type's own
336// operator== must be declared noexcept. This is the recursive guarantee behind
337// operations declared noexcept; a cljonic::Variant declares its own noexcept
338// ==, so detection succeeds through that operator.
339template <typename T>
340inline constexpr bool nothrow_equality_v = requires(const T& left, const T& right) {
341 { left == right } noexcept -> std::convertible_to<bool>;

344// A standard-library range or container is outside the equality domain: `equal`

356// Nominal walker-coverage gate (REQ-NUM-019). The recursive walkers above are
357// enumerative, not reflective: the unspecialized form reports "no floating
358// point / no callable" for every type it does not know. That default is correct
359// only for types whose members are genuinely unanalyzable (a user-defined
360// aggregate, REQ-NUM-018). A library-owned composite exposes its components as
361// template arguments, so its omission from the walkers would silently admit a
362// detectable floating-point comparison.
363//
364// Each library-owned composite therefore declares coverage in its own header.
365// The primary template denies coverage, so an added type that opts into a
366// nominal trait without opting into walker coverage is not stable-equality
367// comparable.
368template <typename T>
369struct walker_coverage {
370 static constexpr bool declared = false;
371};
372
373template <typename T>
374inline constexpr bool is_cljonic_composite_v =
375 is_cljonic_collection_v<T> || is_cljonic_producer_v<T> || is_cljonic_map_entry_v<T> || is_cljonic_variant_v<T>;
376
377template <typename T>
378inline constexpr bool walker_coverage_declared_v = walker_coverage<std::remove_cvref_t<T>>::declared;

380// Types outside the nominal domain keep the benefit of the doubt: their members
381// cannot be analyzed at all, so no coverage declaration is possible or required.
382template <typename T>
383inline constexpr bool walker_coverage_ok_v = !is_cljonic_composite_v<T> || walker_coverage_declared_v<T>;
