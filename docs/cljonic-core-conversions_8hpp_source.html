253template <Floating T>
254[[nodiscard]] constexpr auto binary_magnitude(T value) noexcept -> BinaryMagnitude<T> {
255 int exponent{};
256 T scaled = value;
257 while (scaled >= T{2}) {
258 scaled /= T{2};
259 ++exponent;
260 }
261 while (scaled < T{1}) {
262 scaled *= T{2};
263 --exponent;
264 }
265 return {scaled, exponent};
266}

393template <typename IntegerType>
394[[nodiscard]] constexpr auto distance_between_integers(IntegerType lhs, IntegerType rhs) noexcept -> std::uintmax_t {
395 using I = ValueType<IntegerType>;
396 if constexpr (std::is_signed_v<I>) {
397 const auto lhs_negative = lhs < 0;
398 const auto rhs_negative = rhs < 0;
399 const auto lhs_magnitude = signed_magnitude(lhs);
400 const auto rhs_magnitude = signed_magnitude(rhs);
401 if (lhs_negative != rhs_negative) {
402 return lhs_magnitude + rhs_magnitude;
403 }
404 return unsigned_integer_distance(lhs_magnitude, rhs_magnitude);
405 }
406 return unsigned_integer_distance(static_cast<std::uintmax_t>(lhs), static_cast<std::uintmax_t>(rhs));
407}

418template <typename IntegerType, Floating FloatType>
419[[nodiscard]] constexpr auto integer_distance(IntegerType integer, FloatType floating) noexcept -> std::uintmax_t {
420 using I = ValueType<IntegerType>;
421 const auto candidate = static_cast<long double>(floating);
422 const long double bound = power_of_two(std::numeric_limits<I>::digits);
423
424 if (candidate >= bound) {
425 return distance_to_upper_bound(integer);
426 }
427 if (integer_candidate_below_range<IntegerType>(candidate)) {
428 return std::numeric_limits<std::uintmax_t>::max();
429 }
430
431 return distance_between_integers(integer, static_cast<I>(floating));

434template <Floating T>
435[[nodiscard]] constexpr auto binary_spacing(T value) noexcept -> T {
436 const auto magnitude = binary_magnitude(value);
437 T spacing{1};
438 const int shift = magnitude.exponent - (std::numeric_limits<T>::digits - 1);
439 for (int bit = 0; bit < shift; ++bit) {
440 spacing *= T{2};
441 }
442 for (int bit = 0; bit > shift; --bit) {
443 spacing /= T{2};
444 }
445 return spacing;
446}

453template <Floating T>
454[[nodiscard]] constexpr auto next_positive(T value) noexcept -> T {
455 if (value == std::numeric_limits<T>::max()) {
456 return std::numeric_limits<T>::infinity();
457 }
458 if (value < std::numeric_limits<T>::min()) {
459 if constexpr (std::numeric_limits<T>::has_denorm == std::denorm_present) {
460 return value + std::numeric_limits<T>::denorm_min();
461 } else {
462 return std::numeric_limits<T>::min();
463 }
464 }
465 return value + binary_spacing(value);
466}

468template <Floating T>
469[[nodiscard]] constexpr auto previous_positive(T value) noexcept -> T {
470 if (value <= std::numeric_limits<T>::min()) {
471 if constexpr (std::numeric_limits<T>::has_denorm == std::denorm_present) {
472 return value - std::numeric_limits<T>::denorm_min();
473 } else {
474 return T{0};
475 }
476 }
477 T spacing = binary_spacing(value);
478 if (is_power_of_two(value)) {
479 spacing /= T{2};
480 }
481 return value - spacing;
482}

484template <Floating T>
485[[nodiscard]] constexpr auto next_up(T value) noexcept -> T {
486 if (value < T{0}) {
487 return -previous_positive(-value);
488 }
489 if (value == T{0}) {
490 if constexpr (std::numeric_limits<T>::has_denorm == std::denorm_present) {
491 return std::numeric_limits<T>::denorm_min();
492 } else {
493 return std::numeric_limits<T>::min();
494 }
495 }
496 return next_positive(value);
497}

499template <Floating T>
500[[nodiscard]] constexpr auto next_down(T value) noexcept -> T {
501 if (value > T{0}) {
502 return previous_positive(value);
503 }
504 if (value == T{0}) {
505 if constexpr (std::numeric_limits<T>::has_denorm == std::denorm_present) {
506 return -std::numeric_limits<T>::denorm_min();
507 } else {
508 return -std::numeric_limits<T>::min();
509 }
510 }
511 return -next_positive(-value);
512}

514template <Floating T>
515[[nodiscard]] constexpr auto even_significand(T value) noexcept -> bool {
516 if (value == T{0}) {
517 return true;
518 }
519
520 const T magnitude = value < T{0} ? -value : value;
521 T adjacent = next_up(magnitude);
522 if (!is_finite(adjacent)) {
523 adjacent = previous_positive(magnitude);
524 }
525 const long double spacing = static_cast<long double>(adjacent) - static_cast<long double>(magnitude);
526 const long double significand = static_cast<long double>(magnitude) / spacing;
527 return std::fmod(significand, 2.0L) == 0.0L;
528}

705template <Floating Destination>
706[[nodiscard]] constexpr auto floating_neighbors(Destination candidate, long double value) noexcept
707 -> FloatingNeighbors<Destination> {
708 const auto rounded = static_cast<long double>(candidate);
709 if (rounded < value) {
710 return {candidate, next_up(candidate), false};
711 }
712 if (rounded > value) {
713 return {next_down(candidate), candidate, false};
714 }
715 return {candidate, candidate, true};

718template <Floating Destination>
719[[nodiscard]] constexpr auto nearest_floating_value(Destination lower, Destination upper, long double value) noexcept
720 -> Destination {
721 const long double lower_distance = value - static_cast<long double>(lower);
722 const long double upper_distance = static_cast<long double>(upper) - value;
723 if (lower_distance < upper_distance) {

732template <Floating Destination, Floating SourceType>
733[[nodiscard]] constexpr auto convert_nonzero_floating_nearest(SourceType source, long double value) noexcept
734 -> Result<Destination> {
735 if (!floating_value_in_range<Destination>(value)) {
736 return failure<Destination>(ConversionError::out_of_range);
737 }
738 const auto candidate = static_cast<Destination>(source);
739 const auto neighbors = floating_neighbors(candidate, value);
740 const auto nearest =

743 return failure<Destination>(ConversionError::underflow_to_zero);
744 }
745 return success(nearest);
746}

874template <typename T>
875[[nodiscard]] constexpr auto fallback() noexcept -> T {
876 if constexpr (std::same_as<T, char>) {
877 return '\0';
878 } else if constexpr (std::same_as<T, std::byte>) {
879 return static_cast<std::byte>(UCHAR_MAX);
880 } else {

894template <typename Destination, typename SourceType, typename Policy>
895[[nodiscard]] constexpr auto convert_or_fallback(SourceType source, Policy policy) noexcept -> ValueType<Destination> {
896 const auto result = convert<Destination>(source, policy);
897 if (result.succeeds) {
898 return result.value;
899 }
900
901 if consteval {
902 reject_constant_conversion_failure();
903 } else {
904 runtime_failure(result.error);
905 return fallback<ValueType<Destination>>();
