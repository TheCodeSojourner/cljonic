23 requires(Index < std::tuple_size_v<Tuple>)
24constexpr decltype(auto) invoke_right(const Tuple& functions, Arg&& argument) noexcept {
25 if constexpr (Index == std::tuple_size_v<Tuple> - 1U) {
26 return std::invoke(std::get<Index>(functions), std::forward<Arg>(argument));
27 } else {
28 auto&& right_result = invoke_right<Tuple, Index + 1U>(functions, std::forward<Arg>(argument));
29 return std::invoke(std::get<Index>(functions), std::forward<decltype(right_result)>(right_result));
30 }
