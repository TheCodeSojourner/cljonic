13// Per-position parameter rule (REQ-FN-032): when the callback's parameter types
14// are statically known, each parameter must accept its corresponding source
15// element through a safe implicit conversion. Narrowing conversions and
16// Boolean/character/enumeration/pointer/user-defined crossings are rejected. A
17// callback with fewer parameters than sources is rejected; extra parameters may
18// take defaults.
19template <typename Params, typename IndexSequence, typename... Sources>
20struct MapParametersSafe;

22template <typename Params, std::size_t... Index, typename... Sources>
23struct MapParametersSafe<Params, std::index_sequence<Index...>, Sources...>
24 : std::bool_constant<(concepts::SafeImplicitConversion<std::tuple_element_t<Index, Params>,
25 std::ranges::range_value_t<const Sources>> &&
26 ...)> {};

28template <typename F, typename... Sources>
29inline constexpr bool map_parameters_safe_v = [] constexpr {
30 using Parameters = CallableParameters<std::decay_t<F>>::type;
31 if constexpr (!CallableParameters<std::decay_t<F>>::known) {
32 return true;
33 } else if constexpr (std::tuple_size_v<Parameters> < sizeof...(Sources)) {
34 return false;
35 } else {
36 return MapParametersSafe<Parameters, std::index_sequence_for<Sources...>, Sources...>::value;

38}();

40// Guarded admission for the multi-source `map` free function (REQ-FN-032):
41// every source must be a cljonic source, the transforming function must be
42// copy-constructible and invoked without throwing with one argument per
43// source, its result must be a nothrow-storable element, and -- when its
44// parameter types are inspectable -- every parameter must safely accept the
45// corresponding source element.
46template <typename F, typename... Sources>
47concept MapCallable =
48 (sizeof...(Sources) >= 1) && (concepts::CljonicSource<Sources> && ...) &&
49 std::copy_constructible<std::decay_t<F>> && std::is_nothrow_copy_constructible_v<std::decay_t<F>> &&
50 std::is_nothrow_invocable_v<const std::decay_t<F>&, std::ranges::range_reference_t<const Sources>...> &&
51 concepts::NothrowCollectionElement<
52 std::invoke_result_t<const std::decay_t<F>&, std::ranges::range_reference_t<const Sources>...>> &&
53 map_parameters_safe_v<F, Sources...>;

55template <typename F, typename... Sources>
56using MapResultType = std::invoke_result_t<const std::decay_t<F>&, std::ranges::range_reference_t<const Sources>...>;

58// Compile-time observable-count bound per source: a collection contributes its
59// declared capacity; any producer contributes the configured observable
60// traversal cap (REQ-FN-032 result-capacity policy).
61template <typename Source>
62[[nodiscard]] constexpr auto map_source_bound() noexcept -> std::size_t {
63 if constexpr (concepts::CljonicCollection<Source>) {
64 return Source::capacity();
65 } else {
66 return CLJONIC_COLLECTION_MAXIMUM_ELEMENT_COUNT_VALUE;
67 }
68}

161template <typename F, typename... Sources>
162 requires concepts_detail::MapCallable<F, Sources...>
163[[nodiscard]] constexpr auto map(const F& function, const Sources&... sources) noexcept
164 -> Vector<concepts_detail::MapResultType<F, Sources...>, concepts_detail::MapResultCapacity<Sources...>> {
165 constexpr auto capacity = concepts_detail::MapResultCapacity<Sources...>;
166 using Result = concepts_detail::MapResultType<F, Sources...>;
167
168 const std::size_t mapped_count =
169 std::min({static_cast<std::size_t>(sources.count())..., static_cast<std::size_t>(capacity)});
170
171 auto iterators = std::tuple{sources.begin()...};
172 Vector<Result, capacity> result{};
173 for (std::size_t index = 0U; index < mapped_count; ++index) {
174 const auto value =
175 std::apply([&function](const auto&... iterator) { return std::invoke(function, *iterator...); }, iterators);
176 result = result.conj(value);
177 std::apply([](auto&... iterator) { (++iterator, ...); }, iterators);
178 }
179 return result;
180}
