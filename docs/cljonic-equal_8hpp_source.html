11// The non-cljonic fallthrough domain of `equal` (REQ-FN-002G) is the closed
12// value domain: arithmetic scalars and scoped enums, aggregate-like structs,
13// and cljonic::Variant composites. Standard-library range and container types
14// belong to the C++ interoperability surface, not the equality domain, and are
15// rejected, as is the standard-library variant (it is not a cljonic value
16// type). Unscoped enums and pointers (including member pointers) are also
17// outside the domain: `std::is_scalar_v` would otherwise admit them.
18template <typename T>
19inline constexpr bool in_non_cljonic_fallthrough_domain_v =
20 !is_cljonic_collection_v<T> && !is_cljonic_producer_v<T> && !std::ranges::range<T> &&
21 ((std::is_arithmetic_v<T> || std::is_scoped_enum_v<T>) || std::is_aggregate_v<T> || is_cljonic_map_entry_v<T> ||
22 is_cljonic_variant_v<T> ||
23 // Non-aggregate user-defined classes (REQ-NUM-018, aligned 2026-10-31):
24 // admitted on the stable-equality capability that map keys and set elements
25 // require, together with the non-throwing comparison `equal` itself
26 // requires. Requiring the *comparison* to be non-throwing (not merely
27 // storage) keeps parameterized wrappers whose inner types the library
28 // cannot analyze out of the domain: the standard library does not declare
29 // its wrapper comparisons `noexcept`, so std::pair, std::optional,
30 // std::tuple, and std::variant stay rejected however they are instantiated,
31 // and a floating-point member cannot be smuggled in behind a wrapper.
32 (std::is_class_v<T> && !is_std_variant_v<T> && concepts::NothrowEqualityComparable<T>));
33
34// Equality-family classification for the general-equality free function `equal`
35// (REQ-FN-002G): Vector and Queue join all five producers in the sequential
36// family; Map, Set, and String each form their own family; every other type is
37// outside the family classification.
38template <typename T>
39inline constexpr equality_family equal_family_of_v = [] constexpr -> equality_family {
40 if constexpr (is_cljonic_collection_v<T>) {
41 switch (collection_kind_of_v<T>) {
42 case collection_kind::vector:
43 case collection_kind::queue:
44 return equality_family::sequential;
45 case collection_kind::map:
46 return equality_family::map;
47 case collection_kind::set:
48 return equality_family::set;
49 case collection_kind::string:
50 return equality_family::string;
51 default:
52 return equality_family::none;
53 }
54 } else if constexpr (is_cljonic_producer_v<T>) {
55 return equality_family::sequential;
56 } else {
57 return equality_family::none;
58 }
59}();

104// The value domain admitted by equal's family gates: stable equality whose
105// comparison cannot throw (so the noexcept guarantee cannot be violated) and
106// free of standard-library range components at any depth (equal is not part of
107// the C++ interoperability surface).
108template <typename T>
109concept EqualDomainValue = concepts::NothrowEqualityComparable<T> && !contains_standard_range_v<T>;
110
111// Equality-family gates shared by equal's overloads and not_equal's delegated
112// gate (REQ-FN-002G, REQ-FN-002H); these are not ordering capabilities.
113// Safe cross-type comparison (REQ-FN-002G, human decision 2026-10-31): a pair
114// is comparable when the types are identical or related by a value-preserving
115// implicit conversion in either direction, and the comparison happens in the
116// wider type. Narrowing value relations are harmless under widening; sign
117// crossings, Boolean, character, enumeration, pointer, and user-defined pairs
118// are never safely convertible. Map, Set, String, and Variant alternative
119// identity keep the identical-type rule.
120template <typename Lhs, typename Rhs>
121concept SafelyComparablePair =
122 std::same_as<Lhs, Rhs> || concepts::SafeImplicitConversion<Lhs, Rhs> || concepts::SafeImplicitConversion<Rhs, Lhs>;

124template <typename Lhs, typename Rhs>
125concept EqualScalarPairAdmissible =
126 equal_family_of_v<Lhs> == equality_family::none && equal_family_of_v<Rhs> == equality_family::none &&
127 in_non_cljonic_fallthrough_domain_v<Lhs> && in_non_cljonic_fallthrough_domain_v<Rhs> &&
128 SafelyComparablePair<Lhs, Rhs> && EqualDomainValue<Lhs> && EqualDomainValue<Rhs>;
129
130template <typename Lhs, typename Rhs>
131concept EqualSequentialPairAdmissible =
132 equal_family_of_v<Lhs> == equality_family::sequential && equal_family_of_v<Rhs> == equality_family::sequential &&
133 SafelyComparablePair<equal_value_type_of_t<Lhs>, equal_value_type_of_t<Rhs>> &&
134 EqualDomainValue<equal_value_type_of_t<Lhs>> && EqualDomainValue<equal_value_type_of_t<Rhs>>;
135
136template <typename Lhs, typename Rhs>
137concept EqualStringPairAdmissible =
138 equal_family_of_v<Lhs> == equality_family::string && equal_family_of_v<Rhs> == equality_family::string;

140template <typename Lhs, typename Rhs>
141concept EqualMapPairAdmissible =
142 equal_family_of_v<Lhs> == equality_family::map && equal_family_of_v<Rhs> == equality_family::map &&
143 std::same_as<equal_key_type_of_t<Lhs>, equal_key_type_of_t<Rhs>> &&
144 std::same_as<equal_association_value_type_of_t<Lhs>, equal_association_value_type_of_t<Rhs>> &&
145 EqualDomainValue<equal_key_type_of_t<Lhs>> && EqualDomainValue<equal_association_value_type_of_t<Lhs>>;
146
147template <typename Lhs, typename Rhs>
148concept EqualSetPairAdmissible =
149 equal_family_of_v<Lhs> == equality_family::set && equal_family_of_v<Rhs> == equality_family::set &&
150 std::same_as<equal_value_type_of_t<Lhs>, equal_value_type_of_t<Rhs>> &&
151 EqualDomainValue<equal_value_type_of_t<Lhs>>;

153// Keep every arity on the same family gates so unsupported pairs fail at

156concept EqualPairAdmissible =

173// Lazy element-wise bounded-prefix walk: terminates at the first differing
174// element pair and requires both operands to be exhausted simultaneously for
175// equality. All producers expose cap-bounded begin()/end() traversal, so the
176// walk always terminates, including for two unbounded producers.
177namespace equal_walk_detail {
178
179// Reports whether both iterators have reached their respective ends after the
180// walk, which requires the two sequences to have exhausted simultaneously.
181template <typename LhsIterator, typename RhsIterator, typename LhsEnd, typename RhsEnd>
182[[nodiscard]] constexpr auto both_exhausted(const LhsIterator& lhs_iterator, const LhsEnd& lhs_end,
183 const RhsIterator& rhs_iterator, const RhsEnd& rhs_end) noexcept -> bool {
184 return (lhs_iterator == lhs_end) && (rhs_iterator == rhs_end);
185}
186
187} // namespace equal_walk_detail
188
189template <typename Lhs, typename Rhs>
190[[nodiscard]] constexpr auto equal_prefix_walk(const Lhs& lhs, const Rhs& rhs) noexcept -> bool {
191 using equal_walk_detail::both_exhausted;
192 auto lhs_iterator = lhs.begin();
193 const auto lhs_end = lhs.end();
194 auto rhs_iterator = rhs.begin();
195 const auto rhs_end = rhs.end();
196 while ((lhs_iterator != lhs_end) && (rhs_iterator != rhs_end)) {
197 if (!(*lhs_iterator == *rhs_iterator)) {
198 return false;
199 }
200 ++lhs_iterator;
201 ++rhs_iterator;
202 }
203 return both_exhausted(lhs_iterator, lhs_end, rhs_iterator, rhs_end);
204}

210// Order-insensitive Map matching: every logical entry of the left operand must
211// have a matching key in the right operand whose associated value compares
212// equal (REQ-COLL-021 Map equality semantics).
213template <typename Lhs, typename Rhs>
214[[nodiscard]] constexpr auto map_entries_all_match(const Lhs& lhs, const Rhs& rhs) noexcept -> bool {
215 for (const auto& entry : lhs) {
216 if (!rhs.contains(entry.key) || !(rhs(entry.key) == entry.value)) {
217 return false;
218 }
219 }
220 return true;
221}
222
223// Order-insensitive Set matching: every logical element of the left operand
224// must be present in the right operand (REQ-COLL-021 Set equality semantics).
225template <typename Lhs, typename Rhs>
226[[nodiscard]] constexpr auto set_elements_all_contained(const Lhs& lhs, const Rhs& rhs) noexcept -> bool {
227 for (const auto& element : lhs) {
228 if (!rhs.contains(element)) {
229 return false;
230 }
231 }
232 return true;
233}
