130template <typename F, typename Source>
131 requires concepts_detail::FilterCallable<F, Source>
132[[nodiscard]] constexpr auto filter(const F& predicate, const Source& source) noexcept
133 -> Vector<std::ranges::range_value_t<const Source>, concepts_detail::FilterResultCapacity<Source>> {
134 constexpr auto capacity = concepts_detail::FilterResultCapacity<Source>;
135 using Element = std::ranges::range_value_t<const Source>;
136 const auto observed_count = std::min<std::size_t>(static_cast<std::size_t>(source.count()), capacity);
137 Vector<Element, capacity> result{};
138 auto iterator = source.begin();
139 const auto end = source.end();
140 for (std::size_t index = 0U; index < observed_count && iterator != end; ++index) {
141 const Element element = *iterator;
142 concepts_detail::filter_append_if_matching(predicate, element, result);
143 // Advance only while another observed element remains, so a producer's
144 // step is never advanced past the last element it contributes.
145 if (index + 1U < observed_count) {
146 ++iterator;
147 }
148 }
149 return result;
150}
