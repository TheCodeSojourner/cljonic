75// A single (key, value) pair is admissible for the variadic form when the
76// collection admits it under its own key/value domain (identical to the
77// single-pair overload's gate).
78template <typename C, typename K, typename V>
79concept AssocPairAdmissible =
80 concepts::AssociativeCollection<C> && requires(const C& collection, const K& key, const V& value) {
81 { collection.assoc(key, value) } noexcept -> std::same_as<C>;
82 };
83
84// Every (key, value) pair in the argument list is admissible. The recursion
85// consumes two arguments per step; an empty tail is admissible and a single
86// trailing argument (a key without its value) is not, so an odd argument count
87// can never satisfy the constraint.
88template <typename C, typename... Pairs>
89struct all_assoc_pairs_admissible : std::false_type {};
90
91template <typename C>
92struct all_assoc_pairs_admissible<C> : std::true_type {};
93
94template <typename C, typename K, typename V, typename... Rest>
95struct all_assoc_pairs_admissible<C, K, V, Rest...>
96 : std::bool_constant<AssocPairAdmissible<C, K, V> && all_assoc_pairs_admissible<C, Rest...>::value> {};
97
98template <typename C, typename... Pairs>
99inline constexpr bool all_assoc_pairs_admissible_v = all_assoc_pairs_admissible<C, Pairs...>::value;
100
101// Left-to-right fold of the remaining pairs into an accumulator that already
102// holds the first pair(s); each step reuses the single-pair overload's contract.
103// The accumulator is threaded by reference (never passed by value) so the fold
104// cannot be intercepted by a collection's source-construction constructor.
105template <typename C>
106constexpr void fold_assoc_pairs_into([[maybe_unused]] C& result) noexcept {
107}

109template <typename C, typename K, typename V, typename... Rest>
110constexpr void fold_assoc_pairs_into(C& result, const K& key, const V& value, const Rest&... rest) noexcept {
111 result = result.assoc(key, value);
112 fold_assoc_pairs_into(result, rest...);
113}
